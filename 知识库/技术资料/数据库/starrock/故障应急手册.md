## StarRocks慢查询排查
```sql
--  扫描数据比较大的sql

SELECT *
FROM auditdb.audit_log
WHERE TIME> '2022-09-05 00:00:00'
AND   TIME< '2022-09-05 23:59:59'
AND   scan_rows > 100000000
ORDER BY scan_rows LIMIT 100 \G

-- 超执行时间3ssql查询
SELECT *
FROM auditdb.audit_log
WHERE TIME> CURDATE()
AND   TIME< date_add(curdate(), INTERVAL 1 DAY)
AND   query_time > 3
ORDER BY query_time desc LIMIT 10 \G;

-- 根据查询时间汇总请求量大sql
SELECT regexp_extract(stmt,"sqlId=([^ ]*)",1) AS sqlId,
       CAST(percentile_approx (queryTime,0.99) AS FLOAT) AS P99,
       CAST(percentile_approx (scanRows,0.99) AS FLOAT) AS P99_scan_row,
       AVG(queryTime) AS avgQueryTime,
       MAX(queryTime),
       COUNT(1) AS cnt,
       ROUND(AVG(memCostBytes) / 1024 / 1024,2) AS avgMemCost_M,
       ROUND(AVG(cpuCostNs) / 1000000000,2) AS avgcpuCost_S,
       MAX(returnRows)
FROM auditdb.audit_log
WHERE TIMESTAMP> '2025-04-03 19:00:00'
AND   TIMESTAMP< '2025-04-03 19:10:00'
group  by regexp_extract(stmt,"sqlId=([^ ]*)",1)
ORDER BY cnt DESC 
         LIMIT 10;

-- 根据扫描行数查询sql性能统计
SELECT regexp_extract(stmt,"sqlId=([^ ]*)",1) AS sqlId,
       CAST(percentile_approx (queryTime,0.99) AS FLOAT) AS P99,
       ROUND(AVG(queryTime),2) AS avgQueryTime,
       MAX(queryTime),
       max(scanRows) as scanRows,
       COUNT(1),
       ROUND(AVG(memCostBytes) / 1024 / 1024,2) AS avgMemCost_M,
       ROUND(AVG(cpuCostNs) / 1000000000 ,2) AS avgcpuCost_S,
       MAX(returnRows)
FROM auditdb.audit_log
WHERE TIMESTAMP > '2025-04-01 22:00:00'
AND   TIMESTAMP < '2025-04-01 22:10:00'
group  by regexp_extract(stmt,"sqlId=([^ ]*)",1)
ORDER BY scanRows DESC
         LIMIT 30;  

-- 按并发统计耗时         
SELECT regexp_extract(stmt,"sqlId=([^ ]*)",1) AS sqlId,
       CAST(percentile_approx (queryTime,0.99) AS FLOAT) AS P99,
       ROUND(AVG(queryTime),2) AS avgQueryTime,
       MAX(queryTime),
       MAX(scanRows) as scanRows,
       COUNT(1),
       ROUND(AVG(memCostBytes) / 1024 / 1024,2) AS avgMemCost_M,
       ROUND(AVG(cpuCostNs) / 1000000000 ,2) AS avgcpuCost_S,
       MAX(returnRows)
FROM auditdb.audit_log
WHERE TIMESTAMP > '2025-03-31 22:00:00'
AND   TIMESTAMP < '2025-03-31 22:10:00'
group  by regexp_extract(stmt,"sqlId=([^ ]*)",1)
ORDER BY COUNT(1) DESC
         LIMIT 30;

-- 查询某条sql 组织id 分布情况 
WITH t1 AS
(
  SELECT t.*,
         regexp_extract(stmt,"sqlId=([^ ]*)",1) AS sqlId,
         regexp_replace(regexp_extract(stmt,' t.parent_department_id in \\(([^()]+)\\)',1),' ','') AS pids,
         split(regexp_replace(regexp_extract(stmt,' t.parent_department_id in \\(([^()]+)\\)',1),' ',''),',') as pid_arr,
         ifnull(split_part(regexp_replace(regexp_extract(stmt,' t.parent_department_id in \\(([^()]+)\\)',1),' ',''),',',1),regexp_replace(regexp_extract(stmt,' t.parent_department_id in \\(([^()]+)\\)',1),' ','')) as pid0
  FROM auditdb.audit_log t
WHERE TIMESTAMP> '2025-04-01 20:20:00'
AND   TIMESTAMP< '2025-04-01 20:30:00'
AND   regexp_extract(stmt,"sqlId=([^ ]*)",1) = 'TrunkProblemSummarMapper.getTrunkRealTimeProblemSummary'
),
t2 as (
select t1.*,tdn.hierarchy_code from t1 
inner join dim_coo.dim_coo_department_cur tdn on tdn.department_id = t1.pid0
)
select hierarchy_code,
             count(1),
             CAST(percentile_approx (queryTime,0.99) AS FLOAT) AS P99,
       ROUND(AVG(queryTime),2) AS avgQueryTime,
       MAX(queryTime),       
       ROUND(AVG(memCostBytes) / 1024 / 1024,2) AS avgMemCost_M,
       ROUND(AVG(cpuCostNs) / 1000000000 ,2) AS avgcpuCost_S             
from t2 group by hierarchy_code;
```
