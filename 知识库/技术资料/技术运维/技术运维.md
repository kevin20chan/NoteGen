## kafka运维
```bash
# 查看kafka中topic的配置
kafka-cli admin get topic binlog.pda.transfer

#创建topic
kafka-cli admin create topic tos.depot.batchCode --partitions 64 --replicas 3

## 删除topic
kafka-cli admin delete topic coo.damaged_lose.money_info-uat

# 抓消息
kafkactl consume binlog.pda.pda_route_raw_control --from-beginning | grep KY4000051948749

kafka-cli msg query --from "2025-03-23 15:00:00" --to "2025-03-23 18:07:00" binlog.pda.pda_route_raw_control --show | grep -B 2 KY4000002903701

kafka-cli -C tos msg query --from "2025-06-04 16:35:00" --to "2025-06-04 16:42:00" binlog.pda.pda_route_raw_control --show | grep -B 2 KY4000777596718

kafka-cli msg query --from "2025-12-23 17:00:00" --to "2025-12-23 17:02:00" vms_nearby_driver_into_fence --show -B "vms-prd-public-kafka01.kyeapi.com:9092" -Z "vms-prd-public-kafka01.kyeapi.com:2181"

# 发送消息
kafka-cli produce --partition 0 --value '' binlog.pda.pda_route_raw_control.fix
kafka-cli produce -p 2 binlog.pda.pda_route_raw_control.fix < waybill_22_0628.json

# 抓某个分区某个offset的数据
kafka-cli msg -p 40 --offset 17914 binlog.tms.followFly

# 修改topic分区数(只能增加，不能减少)
kafka-cli admin alter topic binlog.tos.mix --partitions 8

# following
kafka-cli msg openapi.vms.mix --follow

-- 重置kafka消费组offset
kafka-cli admin set offsets binlog.tos.goods --group tos-etl-uat_p4_10_group --allparts --newest
/usr/local/kafka/bin/kafka-consumer-groups.sh  --bootstrap-server 10.201.29.11:9092 --group tos-etl-uat_partitions_128_group --topic binlog.pda.pda_route_raw_control.fix  --reset-offsets --to-latest --execute
```
